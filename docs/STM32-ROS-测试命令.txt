#   W 前进 (0.2m/s)    A 左转    Q 原地左旋 (0.4rad/s)

#   空格 停止     +/- 调速     ESC 退出


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

#         一条命令启动：相机 + 桥接 + RTAB-Map + 键盘遥控
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

cd ~/桌面/Slam && source setup_orbbec_slam.sh
roslaunch orbbec_gemini_slam zystm32_slam.launch

# 弹出窗口：
#   RTAB-Map Viz → 实时 3D 地图（右侧大窗口）
#   RViz         → 相机画面 + 点云
#
# 颜色说明：
#   🔴 红色    = 当前帧实时扫描（正常！）
#   ⬜ 灰白色  = 已建好的地图
#   🟢 绿色    = 回环检测匹配
#   🔵 蓝色    = 回环闭合边


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  建图流程
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  1. 小车放起点，镜头朝前，D435 插 USB3.0 + STM32 USB + 电池
#  2. roslaunch 启动，等两个窗口弹出
#  3. 按 W 缓慢前进，覆盖整个房间
#  4. A/D Q/E 转向，保持平稳
#  5. 走一圈回到起点 → 观察绿色/蓝色回环
#  6. ESC 退出，地图自动保存


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  地图的存储和查看

cp /home/wxd/桌面/Slam/.ros/rtabmap.db /home/wxd/桌面/mymap_$(date +%m%d).db

cd ~/桌面/Slam && source setup_orbbec_slam.sh
rtabmap-databaseViewer /home/wxd/桌面/mymap_151MB.db

rtabmap-info /home/wxd/桌面/mymap_151MB.db
